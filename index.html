    // Recalculate compound financial variables over time

    function updateProjection() {

      const T = state.horizon;

      const R = state.returnTarget / 100.0;

      const PMT = state.monthlyContribution;

      const inflation = 0.025; // 2.5% inflation assumed


      let cashData = [];

      let portfolioData = [];

      let inflationData = [];

      let labels = [];


      for (let y = 0; y <= T; y++) {

        labels.push(`第 ${y} 年`);

        

        // 1. Raw cumulative cash principal saved

        let rawCash = PMT * 12 * y;

        cashData.push(rawCash);


        // 2. SAA Port value with compound interest

        let pfValue = 0;

        if (y > 0) {

          pfValue = PMT * 12 * ((Math.pow(1 + R, y) - 1) / R);

        }

        portfolioData.push(pfValue);


        // 3. Real buying power of principal after 2.5% annual inflation loss

        let infValue = 0;

        if (y > 0) {

          let inflationMultiplier = Math.pow(1 - inflation, y);

          infValue = rawCash * inflationMultiplier;

        } else {

          infValue = 0;

        }

        inflationData.push(infValue);

      }


      // Output final cumulative metrics

      const finalPf = portfolioData[T];

      const finalCash = cashData[T];

      const finalInf = inflationData[T];


      document.getElementById("projPortfolioVal").innerText = `HK$ ${Math.round(finalPf).toLocaleString()}`;

      document.getElementById("projCashVal").innerText = `HK$ ${Math.round(finalCash).toLocaleString()}`;

      document.getElementById("projInflationVal").innerText = `HK$ ${Math.round(finalInf).toLocaleString()}`;


      // Redraw projection canvas chart

      drawChart(labels, portfolioData, inflationData, cashData);

    }

    function drawChart(labels, portfolio, inflation, cash) {

      const canvas = document.getElementById("projectionChart");

      if (!canvas) return;

      const ctx = canvas.getContext("2d");

      

      const width = canvas.clientWidth;

      const height = canvas.clientHeight;

      canvas.width = width * window.devicePixelRatio;

      canvas.height = height * window.devicePixelRatio;

      ctx.scale(window.devicePixelRatio, window.devicePixelRatio);


      ctx.clearRect(0, 0, width, height);


      const padding = { top: 30, right: 30, bottom: 40, left: 75 };

      const chartWidth = width - padding.left - padding.right;

      const chartHeight = height - padding.top - padding.bottom;

      function getX(index) {

        return padding.left + (index / (labels.length - 1)) * chartWidth;

      }

        return padding.top + chartHeight - (value / maxVal) * chartHeight;

      }

        ctx.stroke();

        ctx.font = 'bold 9px monospace';

        ctx.textAlign = 'right';

        ctx.fillText(`$${Math.round(yVal/1000).toLocaleString()}k`, padding.left - 10, yCoord + 3.5);

      }


      // Draw X Axis labels

      ctx.textAlign = 'center';

      const labelStep = Math.max(1, Math.floor(labels.length / 5));

      drawLine(portfolio, '#f59e0b', []);

    // Convert raw Markdown text format from Gemini to beautiful styled HTML block

    function convertMarkdownToHtml(markdownText) {

      if (!markdownText) return "";

      let html = markdownText

        .replace(/&/g, "&amp;")

        .replace(/</g, "&lt;")

        .replace(/>/g, "&gt;");


      // Replace bold syntax **text** with Sun Life brand color bold text

      html = html.replace(/\*\*(.*?)\*\*/g, '<strong class="text-brand-500 font-extrabold">$1</strong>');

      // Replace lists starting with dash or asterisk

      html = html.replace(/^[-\*]\s+(.*?)$/gm, '<li class="ml-4 list-disc text-slate-300 my-1 font-medium">$1</li>');

      // Replace section headers like ### Headers

      html = html.replace(/^###\s+(.*?)$/gm, '<h3 class="text-xs sm:text-sm font-black text-white uppercase tracking-widest mt-4 mb-2 flex items-center border-l-3 border-brand-500 pl-2">$1</h3>');

      // Replace new lines with breaks

      html = html.split('\n').join('<br>');


      return html;

    }


    // Custom Token prioritized API call logic with robust Exponential Backoff (Up to 5 Retries)

    async function fetchGeminiReview(systemInstruction, userPrompt) {

      const apiKey = ""; // Runtime automatically provisions this API key

      const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

      

      const payload = {

        contents: [

          {

            parts: [{ text: userPrompt }]

          }

        ],

        systemInstruction: {

          parts: [{ text: systemInstruction }]

        }

      };


      let currentDelay = 1000; // 1s start delay

      for (let attempt = 1; attempt <= 5; attempt++) {

        try {

          const response = await fetch(url, {

            method: "POST",

            headers: {

              "Content-Type": "application/json"

            },

            body: JSON.stringify(payload)

          });


          if (response.ok) {

            const result = await response.json();

            const textResponse = result.candidates?.[0]?.content?.parts?.[0]?.text;

            if (textResponse) {

              return textResponse;

            }

          }

        } catch (error) {

          // Do not log retry failures to terminal console as per instructions

        }


        if (attempt < 5) {

          await new Promise(resolve => setTimeout(resolve, currentDelay));

          currentDelay *= 2; // Double delay on each backoff cycle (1s, 2s, 4s, 8s, 16s)

        }

      }


      throw new Error("無法連接至 AI 投顧智能服務。請確認您已開啟開發人員接口，或稍後再試。");

    }
